cmake_minimum_required(VERSION 3.24)
project(NeonDebugger LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_WIN32_EXECUTABLE ON)

include(FetchContent)
FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.7.tar.gz)
FetchContent_MakeAvailable(lua)

file(GLOB LUA_SOURCES "${lua_SOURCE_DIR}/src/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX ".*/lua.c$")
list(FILTER LUA_SOURCES EXCLUDE REGEX ".*/luac.c$")
add_library(lua_static STATIC ${LUA_SOURCES})
target_include_directories(lua_static PUBLIC "${lua_SOURCE_DIR}/src")
target_compile_definitions(lua_static PRIVATE LUA_USE_WINDOWS _CRT_SECURE_NO_WARNINGS)

add_executable(NeonDebugger WIN32 src/main.cpp)
target_include_directories(NeonDebugger PRIVATE "${lua_SOURCE_DIR}/src")
target_link_libraries(NeonDebugger PRIVATE lua_static psapi comctl32)
target_compile_definitions(NeonDebugger PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS NOMINMAX)

if(MSVC)
  target_compile_options(NeonDebugger PRIVATE /W4 /permissive- /MT)
  target_compile_options(lua_static PRIVATE /MT)
else()
  target_compile_options(NeonDebugger PRIVATE -Wall -Wextra)
endif()

install(TARGETS NeonDebugger RUNTIME DESTINATION .)
